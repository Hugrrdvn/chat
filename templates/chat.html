import os import sqlite3 from datetime import datetime from flask import Flask, request, jsonify from flask_socketio import SocketIO, emit from flask_cors import CORS # ===================================== # ROYAL CHAT SERVER # ===================================== app = Flask(__name__) app.config["SECRET_KEY"] = os.environ.get( "SECRET_KEY", "royal-chat-secret-key" ) CORS(app) socketio = SocketIO( app, cors_allowed_origins="*", async_mode="threading" ) DATABASE = "royal_chat.db" # ===================================== # DATABASE # ===================================== def get_db(): connection = sqlite3.connect(DATABASE) connection.row_factory = sqlite3.Row return connection def initialize_database(): with get_db() as db: db.execute(""" CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL, avatar TEXT, message TEXT NOT NULL, message_time TEXT NOT NULL ) """) db.commit() initialize_database() # ===================================== # ONLINE USERS # ===================================== online_users = {} # ===================================== # CHAT HISTORY # ===================================== def get_message_history(limit=100): with get_db() as db: rows = db.execute(""" SELECT id, username, avatar, message, message_time FROM messages ORDER BY id DESC LIMIT ? """, (limit,)).fetchall() return [ { "id": row["id"], "username": row["username"], "avatar": row["avatar"], "text": row["message"], "time": row["message_time"] } for row in reversed(rows) ] # ===================================== # SAVE MESSAGE # ===================================== def save_message(username, avatar, message): message_time = datetime.now().strftime("%H:%M") with get_db() as db: cursor = db.execute(""" INSERT INTO messages (username, avatar, message, message_time) VALUES (?, ?, ?, ?) """, ( username, avatar, message, message_time )) db.commit() message_id = cursor.lastrowid return { "id": message_id, "username": username, "avatar": avatar, "text": message, "time": message_time } # ===================================== # ONLINE COUNT # ===================================== def broadcast_online_count(): count = len(online_users) socketio.emit( "update_user_count", count, to=None ) # ===================================== # USER JOIN # ===================================== @socketio.on("join_chat") def join_chat(data): username = str( data.get("username", "عضو") ).strip()[:50] avatar = str( data.get("avatar", "") ).strip()[:500] if not username: return online_users[request.sid] = { "username": username, "avatar": avatar } emit( "load_history", get_message_history() ) broadcast_online_count() emit( "system_notification", { "username": username, "text": "انضم إلى الجروب" }, broadcast=True, include_self=False ) # ===================================== # RECEIVE AND BROADCAST MESSAGE # ===================================== @socketio.on("send_message") def send_message(data): if request.sid not in online_users: return user = online_users[request.sid] message = str( data.get("text", "") ).strip() if not message: return if len(message) > 4000: return saved_message = save_message( username=user["username"], avatar=user["avatar"], message=message ) socketio.emit( "receive_message", saved_message, to=None ) # ===================================== # EDIT MESSAGE # ===================================== @socketio.on("edit_message") def edit_message(data): if request.sid not in online_users: return message_id = data.get("id") new_text = str(data.get("text", "")).strip() if not message_id or not new_text: return if len(new_text) > 4000: return username = online_users[request.sid]["username"] with get_db() as db: result = db.execute(""" UPDATE messages SET message = ? WHERE id = ? AND username = ? """, ( new_text, message_id, username )) db.commit() if result.rowcount == 0: return socketio.emit( "message_edited", { "id": message_id, "text": new_text }, to=None ) # ===================================== # DELETE MESSAGE # ===================================== @socketio.on("delete_message") def delete_message(data): if request.sid not in online_users: return message_id = data.get("id") if not message_id: return username = online_users[request.sid]["username"] with get_db() as db: result = db.execute(""" DELETE FROM messages WHERE id = ? AND username = ? """, ( message_id, username )) db.commit() if result.rowcount == 0: return socketio.emit( "message_deleted", {"id": message_id}, to=None ) # ===================================== # TYPING STATUS # ===================================== @socketio.on("typing") def typing(data): if request.sid not in online_users: return emit( "user_typing", { "username": online_users[request.sid]["username"], "typing": bool(data.get("typing", False)) }, broadcast=True, include_self=False ) # ===================================== # USER DISCONNECT # ===================================== @socketio.on("disconnect") def disconnect(): user = online_users.pop(request.sid, None) broadcast_online_count() if user: emit( "system_notification", { "username": user["username"], "text": "غادر الجروب" }, broadcast=True ) # ===================================== # MEMBERS API # ===================================== @app.route("/api/members") def members(): return jsonify({ "count": len(online_users), "members": list(online_users.values()) }) # ===================================== # HISTORY API # ===================================== @app.route("/api/history") def history(): return jsonify({ "messages": get_message_history() }) # ===================================== # SERVER STATUS # ===================================== @app.route("/api/status") def status(): return jsonify({ "server": "Royal Chat", "status": "online", "online_users": len(online_users), "database": "connected", "messaging": "enabled" }) # ===================================== # HOME # ===================================== @app.route("/") def home(): return """ Royal Chat Server

👑 Royal Chat

السيرفر يعمل بنجاح

نظام استقبال وإرسال الرسائل نشط

قاعدة البيانات متصلة

Server Status """ # ===================================== # START SERVER # ===================================== if __name__ == "__main__": port = int(os.environ.get("PORT", 5000)) socketio.run( app, host="0.0.0.0", port=port, debug=False, allow_unsafe_werkzeug=True )
