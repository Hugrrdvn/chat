import os import sqlite3 from datetime import datetime from threading import Lock from flask import Flask, jsonify from flask_socketio import SocketIO, emit from flask_cors import CORS # ========================================== # ROYAL CHAT - REAL-TIME CHAT SERVER # ========================================== app = Flask(__name__) app.config["SECRET_KEY"] = os.environ.get( "SECRET_KEY", "royal-chat-production-secret" ) CORS(app) socketio = SocketIO( app, cors_allowed_origins="*", async_mode="threading" ) DATABASE = os.path.join( os.path.dirname(os.path.abspath(__file__)), "royal_chat.db" ) users_lock = Lock() online_users = {} # ========================================== # DATABASE # ========================================== def get_db(): db = sqlite3.connect( DATABASE, timeout=30 ) db.row_factory = sqlite3.Row return db def initialize_database(): with get_db() as db: db.execute(""" CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT NOT NULL, avatar TEXT, message TEXT NOT NULL, message_time TEXT NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) """) db.commit() initialize_database() # ========================================== # HELPERS # ========================================== def get_current_user(): with users_lock: return online_users.get(request_sid()) def request_sid(): from flask import request return request.sid def get_history(limit=100): with get_db() as db: rows = db.execute(""" SELECT id, username, avatar, message, message_time FROM messages ORDER BY id DESC LIMIT ? """, (limit,)).fetchall() return [ { "id": row["id"], "username": row["username"], "avatar": row["avatar"] or "", "text": row["message"], "time": row["message_time"] } for row in reversed(rows) ] def save_message(username, avatar, text): message_time = datetime.now().strftime("%H:%M") with get_db() as db: cursor = db.execute(""" INSERT INTO messages ( username, avatar, message, message_time ) VALUES (?, ?, ?, ?) """, ( username, avatar, text, message_time )) db.commit() message_id = cursor.lastrowid return { "id": message_id, "username": username, "avatar": avatar, "text": text, "time": message_time } def broadcast_online_count(): with users_lock: count = len(online_users) socketio.emit( "update_user_count", count ) # ========================================== # USER JOIN # ========================================== @socketio.on("join_chat") def join_chat(data): if not isinstance(data, dict): return username = str( data.get("username", "") ).strip()[:50] avatar = str( data.get("avatar", "") ).strip()[:1000] if not username: return with users_lock: online_users[request_sid()] = { "username": username, "avatar": avatar } emit( "load_history", get_history() ) broadcast_online_count() socketio.emit( "system_notification", { "username": username, "text": "انضم إلى الجروب" }, skip_sid=request_sid() ) # ========================================== # SEND REAL MESSAGE # ========================================== @socketio.on("send_message") def send_message(data): if not isinstance(data, dict): return user = get_current_user() if not user: return text = str( data.get("text", "") ).strip() if not text or len(text) > 4000: return message = save_message( username=user["username"], avatar=user["avatar"], text=text ) socketio.emit( "receive_message", message ) # ========================================== # EDIT MESSAGE # ========================================== @socketio.on("edit_message") def edit_message(data): if not isinstance(data, dict): return user = get_current_user() if not user: return message_id = data.get("id") new_text = str( data.get("text", "") ).strip() if not message_id or not new_text: return if len(new_text) > 4000: return with get_db() as db: result = db.execute(""" UPDATE messages SET message = ? WHERE id = ? AND username = ? """, ( new_text, message_id, user["username"] )) db.commit() if result.rowcount == 0: return socketio.emit( "message_edited", { "id": message_id, "text": new_text } ) # ========================================== # DELETE MESSAGE # ========================================== @socketio.on("delete_message") def delete_message(data): if not isinstance(data, dict): return user = get_current_user() if not user: return message_id = data.get("id") if not message_id: return with get_db() as db: result = db.execute(""" DELETE FROM messages WHERE id = ? AND username = ? """, ( message_id, user["username"] )) db.commit() if result.rowcount == 0: return socketio.emit( "message_deleted", {"id": message_id} ) # ========================================== # TYPING INDICATOR # ========================================== @socketio.on("typing") def typing(data): user = get_current_user() if not user: return emit( "user_typing", { "username": user["username"], "typing": bool(data.get("typing", False)) }, broadcast=True, include_self=False ) # ========================================== # DISCONNECT # ========================================== @socketio.on("disconnect") def disconnect(): sid = request_sid() with users_lock: user = online_users.pop(sid, None) broadcast_online_count() if user: socketio.emit( "system_notification", { "username": user["username"], "text": "غادر الجروب" } ) # ========================================== # ONLINE MEMBERS API # ========================================== @app.route("/api/members") def members(): with users_lock: users = list(online_users.values()) return jsonify({ "count": len(users), "members": users }) # ========================================== # MESSAGE HISTORY API # ========================================== @app.route("/api/history") def history(): return jsonify({ "messages": get_history(200) }) # ========================================== # SERVER STATUS # ========================================== @app.route("/api/status") def status(): with users_lock: count = len(online_users) return jsonify({ "server": "Royal Chat", "status": "online", "online_users": count, "database": "connected", "messaging": "active" }) # ========================================== # HEALTH CHECK # ========================================== @app.route("/") def home(): return jsonify({ "name": "Royal Chat", "status": "running", "socketio": "enabled", "database": "connected" }) # ========================================== # START SERVER # ========================================== if __name__ == "__main__": port = int( os.environ.get("PORT", 5000) ) socketio.run( app, host="0.0.0.0", port=port, debug=False, allow_unsafe_werkzeug=True )
